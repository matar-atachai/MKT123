ตั้งค่าเชื่อมต่อ Google Sheet

รับข้อมูลออเดอร์ ชื่อ ที่อยู่ เบอร์ เข้าสเปรดชีตของคุณ

หากปล่อยว่าง ระบบจะจำลองการส่ง และจัดเก็บในคลังประวัติของหน้านี้

โค้ด Google Apps Script (ก๊อปปี้ไปวางได้เลย) 
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["วันที่-เวลา", "รหัสออเดอร์", "ชื่อช่าง/ร้าน", "เบอร์โทร", "ที่อยู่หน้างาน", "วิธีจัดส่ง", "รายการอะไหล่", "ยอดรวม (บาท)", "หมายเหตุ"]);
    }
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      new Date(),
      data.orderId || "",
      data.name || "",
      data.phone || "",
      data.address || "",
      data.deliveryMethod || "",
      data.items || "",
      data.total || 0,
      data.notes || ""
    ]);
    return ContentService.createTextOutput(JSON.stringify({status: 'success'})).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: 'error', message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

วิธีติดตั้งแบบง่ายใน 1 นาที:

  1. เปิด Google Sheets ใหม่
  2. ไปที่ ส่วนขยาย (Extensions) > Apps Script
  3. ลบโค้ดเดิม แล้ววางโค้ดด้านบนลงไป แล้วกดปุ่ม บันทึก (Save)
  4. กด ทำให้ใช้งานได้ (Deploy) > การทำให้ใช้งานได้ใหม่ (New deployment)
  5. เลือกประเภท เว็บแอป (Web app), ตั้งค่า ผู้มีสิทธิ์เข้าถึง (Who has access) เป็น ทุกคน (Anyone) แล้วกด Deploy นำ URL มาใส่ในช่องด้านบน
